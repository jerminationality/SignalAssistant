import QtQuick

/**
 * BatchedFretboardBinOverlay - High-performance canvas-based heatmap overlay
 * 
 * Renders 150 fretboard bins (6 strings × 25 frets) using immediate-mode canvas.
 * Uses Pi5-stable rendering strategy with SVG coordinate mapping.
 */
Canvas {
    id: root
    
    // ========================================================================
    // REQUIRED PROPERTIES
    // ========================================================================
    
    // Parent must provide these function properties:
    // - colorProvider(stringIdx1to6, fret0to24) -> color
    // - strokeProvider(stringIdx1to6, fret0to24) -> color
    
    required property var colorProvider
    required property var strokeProvider
    
    // ========================================================================
    // CONFIGURATION
    // ========================================================================
    
    // SVG coordinate system (matching Fret Cells.svg viewBox: 0 0 1216 127)
    readonly property var fretX: [
        0, 48.5, 95, 139, 181, 221, 259, 295, 329, 361, 391, 419, 
        445, 469, 491, 511, 529, 545, 559, 571, 581, 589, 595, 599, 601
    ]
    
    readonly property var fretW: [
        48, 46, 44, 42, 40, 38, 36, 34, 32, 30, 28, 26,
        24, 22, 20, 18, 16, 14, 12, 10, 8, 6, 4, 2, 1
    ]
    
    readonly property var stringY: [
        10.5,   // String 0 (Low E)
        30.5,   // String 1 (A)
        50.5,   // String 2 (D)
        70.5,   // String 3 (G)
        90.5,   // String 4 (B)
        110.5   // String 5 (High e)
    ]
    
    readonly property real stringH: 16.0
    
    // ========================================================================
    // RENDERING
    // ========================================================================
    
    // CRITICAL: Use Canvas.Immediate for Pi5 stability
    // Threaded rendering causes race conditions and visual artifacts
    renderStrategy: Canvas.Immediate
    renderTarget: Canvas.FramebufferObject
    
    // Anti-aliasing (optional, may impact performance)
    antialiasing: true
    
    // ========================================================================
    // PAINT HANDLER
    // ========================================================================
    
    onPaint: {
        const ctx = getContext("2d");
        if (!ctx) return;
        
        // Clear canvas
        ctx.clearRect(0, 0, width, height);
        
        // Calculate scale factors (canvas size vs SVG viewBox)
        const scaleX = width / 1216.0;
        const scaleY = height / 127.0;
        
        // Draw all 150 bins (6 strings × 25 frets)
        for (let s = 0; s < 6; ++s) {
            const y = stringY[s] * scaleY;
            const h = stringH * scaleY;
            
            for (let f = 0; f < 25; ++f) {
                const x = fretX[f] * scaleX;
                const w = fretW[f] * scaleX;
                
                // Get colors from parent providers (1-based string index)
                const fillColor = colorProvider(s + 1, f);
                const strokeColor = strokeProvider(s + 1, f);
                
                // Draw filled rectangle
                ctx.fillStyle = fillColor;
                ctx.fillRect(x, y, w, h);
                
                // Draw stroke (0.5px for subtle borders)
                ctx.strokeStyle = strokeColor;
                ctx.lineWidth = 0.5;
                ctx.strokeRect(x, y, w, h);
            }
        }
    }
    
    // ========================================================================
    // PUBLIC API
    // ========================================================================
    
    /**
     * Request a repaint (called when bin magnitudes change)
     */
    function requestPaint() {
        root.requestPaint();
    }
    
    // ========================================================================
    // AUTO-REPAINT ON SIZE CHANGE
    // ========================================================================
    
    onWidthChanged: requestPaint()
    onHeightChanged: requestPaint()
}

// BIN MAGNITUDE CACHE UPDATE (10Hz TIMER)
// ============================================================================

void TabEngineBridge::updateBinMagnitudeCache() {
    if (!m_atomicState) {
        return;
    }
    
    // Check if frame counter advanced (new CQT data available)
    const quint64 currentFrame = m_atomicState->frameCounter();
    if (currentFrame == m_lastFrameCounter) {
        return;  // No new data
    }
    m_lastFrameCounter = currentFrame;
    
    // Read all 150 bin magnitudes from AtomicNoteState
    bool anyChange = false;
    for (int s = 0; s < 6; ++s) {
        for (int f = 0; f < 25; ++f) {
            const float magnitude = m_atomicState->getBinMagnitude(s, f);
            
            // Update cache if changed (gate at 0.005 to reduce noise)
            if (std::abs(magnitude - m_binMagnitudeCache[s][f]) > 0.005f) {
                m_binMagnitudeCache[s][f] = magnitude;
                anyChange = true;
            }
            
            // Update threshold cache (using physics-based model)
            const float envFloor = m_store->activeValue(NoteParameter::EnvelopeFloor, s);
            const float thresholdMultiplier = CQTNoteDetector::getThresholdMultiplier(s, f);
            m_binThresholdCache[s][f] = envFloor * thresholdMultiplier;
        }
    }
    
    // Log magnitude update to HeatmapLogger
    HeatmapLogger::instance().logMagnitudeUpdate(
        currentFrame,
        m_binMagnitudeCache,
        m_binThresholdCache
    );
    
    // Emit signal if any bin changed
    if (anyChange) {
        emit binMagnitudesChanged();
    }
}
